无需返工任何业务代码（`src/` 与 `m5-hotfix-cocreation.suite.js` 均已处于完美封板状态），仅需在执行 GitHub 推送前完成以下 3 步工程同步：

### Step 1：将 `compactJS` 正式落盘至 `tests/tools/build-static-bundle.mjs`

在 [tests/tools/build-static-bundle.mjs](https://drive.google.com/file/d/[REDACTED]/view?usp=drivesdk) 的 `writeFileSync(outFile, bundle, 'utf8')` 前合入以下零依赖词法压缩函数（同时统一换行符为 `\n` 以消除 `105 KB` 的 CRLF 膨胀，使 Windows 下构建体积也稳定在 `1.88 MiB`）：

```js
function compactJS(source) {
  const s = source.replace(/\r\n/g, '\n');
  let out = '';
  let i = 0;
  const n = s.length;
  let state = 'CODE'; // CODE | SL_COMMENT | ML_COMMENT | SQUOTE | DQUOTE | TEMPLATE | REGEX
  let lastSigChar = '\n';

  const canPrecedeRegex = (ch) =>
    '(,=:[!&|?+{};~\n'.includes(ch);

  while (i < n) {
    const c = s[i];
    const next = s[i + 1] || '';

    if (state === 'CODE') {
      if (c === '/' && next === '/') {
        state = 'SL_COMMENT';
        i += 2;
        continue;
      }
      if (c === '/' && next === '*') {
        state = 'ML_COMMENT';
        i += 2;
        continue;
      }
      if (c === "'") {
        state = 'SQUOTE';
        out += c;
        lastSigChar = c;
        i++;
        continue;
      }
      if (c === '"') {
        state = 'DQUOTE';
        out += c;
        lastSigChar = c;
        i++;
        continue;
      }
      if (c === '`') {
        state = 'TEMPLATE';
        out += c;
        lastSigChar = c;
        i++;
        continue;
      }
      if (c === '/' && canPrecedeRegex(lastSigChar)) {
        state = 'REGEX';
        out += c;
        lastSigChar = c;
        i++;
        continue;
      }
      if (c === '\n') {
        if (out.length > 0 && out[out.length - 1] !== '\n') {
          out += '\n';
        }
        i++;
        while (i < n && (s[i] === ' ' || s[i] === '\t' || s[i] === '\n')) i++;
        continue;
      }
      if (c === ' ' || c === '\t') {
        if (out.length > 0 && out[out.length - 1] !== ' ' && out[out.length - 1] !== '\n') {
          out += ' ';
        }
        i++;
        continue;
      }
      out += c;
      lastSigChar = c;
      i++;
    } else if (state === 'SL_COMMENT') {
      if (c === '\n') {
        state = 'CODE';
        if (out.length > 0 && out[out.length - 1] !== '\n') out += '\n';
      }
      i++;
    } else if (state === 'ML_COMMENT') {
      if (c === '*' && next === '/') {
        state = 'CODE';
        i += 2;
      } else {
        i++;
      }
    } else if (state === 'SQUOTE') {
      out += c;
      if (c === '\\' && i + 1 < n) {
        out += s[i + 1];
        i += 2;
        continue;
      }
      if (c === "'") state = 'CODE';
      i++;
    } else if (state === 'DQUOTE') {
      out += c;
      if (c === '\\' && i + 1 < n) {
        out += s[i + 1];
        i += 2;
        continue;
      }
      if (c === '"') state = 'CODE';
      i++;
    } else if (state === 'TEMPLATE') {
      out += c;
      if (c === '\\' && i + 1 < n) {
        out += s[i + 1];
        i += 2;
        continue;
      }
      if (c === '`') state = 'CODE';
      i++;
    } else if (state === 'REGEX') {
      out += c;
      if (c === '\\' && i + 1 < n) {
        out += s[i + 1];
        i += 2;
        continue;
      }
      if (c === '/' || c === '\n') state = 'CODE';
      i++;
    }
  }
  return out;
}
```
