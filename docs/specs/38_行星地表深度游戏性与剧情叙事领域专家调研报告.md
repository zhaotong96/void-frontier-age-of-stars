零外部资产依赖 + GPU 实例化批处理”的架构特征，专家组从上述对标库中甄选出改造成本最低、玩法杠杆率最高、对 WebGL 帧率零冲击的三大核心机制进行工程移植。第一项移植机制是源自《Outer Wilds》与《No Man's Sky》的基于余弦锥角与反平方衰减的“多光谱信号镜与三维测向仪”。在《Outer Wilds》中，玩家无需依赖满屏杂乱的图标，而是举起信号镜（Signalscope）旋转视角，根据音频脉冲频率与波形振幅寻找未知信标。移植至《无垠边疆》后，当玩家按下 KeyV 开启战术分析面罩时，可通过 Tab 键在【生态光谱（Bio-Spectrum）】、【电磁测向（EM-Signalscope）】与【地质穿透（Geo-Penetrator）】三种模式间切换。在电磁测向模式下，设玩家相机水平朝向单位向量为 $\mathbf{f} = (-\sin\psi, -\cos\psi)$，角色指向第 $k$ 个隐藏信号源（SOS 求救信标、失联前哨或先驱者谐振碑）的水平向量为 $\mathbf{d}_k = \mathbf{p}_k - \mathbf{p}_{\text{char}}$，距离 $r_k = \|\mathbf{d}_k\|$，夹角余弦 $\cos\theta_k = \frac{\mathbf{f} \cdot \mathbf{d}_k}{r_k}$。该信号源对目镜接收器的瞬时信噪比贡献 $S_k$ 遵循定向高斯波束与距离衰减公式：$$S_k(r_k, \theta_k) = I_0 \cdot \exp\left(-\frac{r_k}{R_{\text{atten}}}\right) \cdot \max\left(0, \cos\theta_k\right)^{\gamma} \cdot \left(1 - \eta_{\text{storm}}\right)$$式中 $I_0 = 1.0$ 为信标发射强度，$R_{\text{atten}}$（取值 180 m）为特征衰减距离，$\gamma = 6.0$ 为目镜定向聚焦指数（当准星偏离目标超过 45° 时信号迅速衰减为背景白噪声），$\eta_{\text{storm}}$（取值范围 0–0.75）为电磁磁暴（em_plasma_superstorm）期间的电离干扰系数。在工程改造成本上，该机制实现了零额外 DrawCall 与零外部音频依赖：波形渲染完全复用 GroundView._drawHudFree 现有的 2D Canvas 正弦波绘制通道（原代码已在纵坐标 hy + 46 处绘制静态装饰声谱波），只需将正弦波振幅与频率绑定至实时计算的 $\sum_k S_k$，并在 PlanetTerrainMesh.js 中复用现有的 uFootsteps 环形缓冲区第 7 槽位发射指向信号源的脉冲波纹；同时直接调用现有 AudioManager 的 Web Audio API 振荡器根据 $S_k$ 动态调制合成音脉冲间隔（从远距 1.5 s/次的低频滴答声加速至近距 0.12 s/次的高频锁定蜂鸣），仅需在 GroundView.js 扩展约 65 行代码，0 新增 Shader，0 新增 GL Buffer。第二项移植机制是源自《Helldivers 2》与《Subnautica》的单 VAO 部件状态掩码驱动的“地表微副本与终端微解谜”。传统 3D 游戏制作废墟内部探索需要数万面的室内建模与复杂碰撞体，在轻量级 WebGL 中极易突破 15,000 面预算并引发弹簧臂相机穿模。我们将每个地表 POI 设计为“局部 SDF 平整凹坑工事 + 3–4 个模块化可交互子部件（气密门、供能导线盒、加密科学终端、核心物资舱）”的露天或半掩埋紧凑微副本。在 WebGL 渲染实现上，当前 GroundView.js 的 BuildingMeshBatch 已通过逐顶点属性 aMesh 与实例属性 iParam.y（网格槽位号）、iParam.z（自发光 vE）、iParam.w（受击闪白 vF）实现了 1 次 DrawCall 渲染全部 21 类建筑与地标。我们仅需将 BuildingMeshBatch 的容量 this.cap 从 384 提升至 512，并在图集尾部追加 12 个 POI 预制件槽位与 4 个野战部署物槽位。当玩家消耗外骨骼能量用激光切割开坠毁残骸的气密门（阶段 1），或完成 WASD/方向键频段对齐解谜重启前哨电源（阶段 2）时，无需重建任何 WebGL VAO，只需在每帧组装 _bmeshList() 实例数组时根据 POI 的 solvedMask 位掩码动态修改对应子部件实例的 emit（通电后从 0.0 变为 1.35 青绿辉光）与 rotY/scale（气密门旋转开启或坍缩为已发掘基座），即可在 0 额外 DrawCall 下实现高沉浸感的微副本解谜流转。第三项移植机制是源自《No Man's Sky》的基于连续微分热度计与 GPU 顶点代数动画的“生态感知与 1–5 星哨兵响应”。为实现野生动物受惊奔逃、掠食者夜间伏击以及 1–5 星哨兵增援，我们升级 BioFloraPass.js 通道 B（FAUNA_VS）与 export const SOS_STATE = Object.freeze({ INACTIVE: 0, TRACKING: 1, TRIGGERED: 2, RESOLVED: 3 });

/**
 * 惰性初始化行星地表深度状态容器 planet.surfaceDelta（仅在玩家首次登陆该行星时调用）
 */
export function ensureSurfaceDelta(world, planet) {
  if (planet.surfaceDelta && planet.surfaceDelta.v === 1) return planet.surfaceDelta;
  const seeds = derivePlanetSeedsExtended(world, planet.id);
  const sosSeed = mix32(seeds.planetSeed ^ 0x50550501);
  // 75% 的可登陆非气态行星生成一条加密求救信标
  const hasSos = planet.category !== 'gasGiant' && ((sosSeed % 100) < 75);
  const ang = ((sosSeed >>> 8) % 3600) / 3600 * Math.PI * 2;
  const rad = 145 + ((sosSeed >>> 20) % 85); // 位于 145m..230m 外围环带
  const branch = (sosSeed >>> 4) % 3;       // 0:幸存者救援, 1:全灭遗言日志, 2:海盗伏击陷阱

  planet.surfaceDelta = {
    v: 1,
    // sos 5元组: [state, srcX_dm, srcZ_dm, branchKind, resolvedChoice]
    sos: hasSos ? [SOS_STATE.TRACKING, Math.round(Math.cos(ang) * rad * 10), Math.round(Math.sin(ang) * rad * 10), branch, 0] : null,
    // extraction 5元组: [active(0|1), x_dm, z_dm, remainSec_d10, completedCount]
    extraction: [0, 0, 0, 0, 0],
    // heat 4元组: [heatPermille(0..1000), starLevel(0..5), evadeSec_d10, bastionDefeated(0|1)]
    heat: [0, 0, 0, 0],
    // 玩家部署在野外的便携装备列表（上限 8 个，每个为 6 元组: [id, kindIdx, x_dm, z_dm, hp, ttlSec]）
    deployables: [],
    // 已收集的数据板日志位掩码（bit 0..23 对应 8 组剧本 × 3 则日志）
    slatesMask: 0,
    // 已扫描的特殊植物掩码与地质奇观标志
    floraScannedMask: 0,
    geoWonderScanned: 0,
  };
  return planet.surfaceDelta;
}

/**
 * 1. 紧急求救信号方位测向仪计算（SOS Triangulation — 对标 Outer Wilds Signalscope）
 * 根据玩家当前世界坐标 (charX, charZ) 与相机朝向 yaw，计算信号强度 (0..1) 与杂音干扰度 (0..1)
 */
export function evaluateSosSignalscope(planet, charX, charZ, camYaw, stormEmFactor = 0) {
  const sd = planet?.surfaceDelta;
  if (!sd || !sd.sos || sd.sos[0] !== SOS_STATE.TRACKING) {
    return { active: false, signalStrength: 0, noiseLevel: 1, distM: Infinity, bearingDiffRad: Math.PI };
  }
  const sx = sd.sos[1] / 10;
  const sz = sd.sos[2] / 10;
  const dx = sx - charX;
  const dz = sz - charZ;
  const distM = Math.hypot(dx, dz);

  // 相机前向向量 (-sin(yaw), -cos(yaw)) 与目标方向夹角
  const fx = -Math.sin(camYaw);
  const fz = -Math.cos(camYaw);
  const cosTheta = distM > 1e-4 ? (fx * dx + fz * dz) / distM : 1.0;
  const directionalGain = Math.pow(Math.max(0, cosTheta), 6.0); // 6次方波束聚焦
  const distanceAtten = Math.exp(-distM / 180.0);

  const rawSignal = clamp(distanceAtten * (0.20 + 0.80 * directionalGain) * (1.0 - 0.70 * stormEmFactor), 0, 1);
  const noiseLevel = clamp(1.0 - rawSignal + 0.35 * stormEmFactor, 0.05, 1.0);
  return { active: true, signalStrength: rawSignal, noiseLevel, distM, cosTheta };
}

/**
 * 抵达 SOS 信标源头（距离 <= 6.0m 按 E 键或自动触发）：解算三大结局分支
 */
export function triggerSosBeaconOutcome(world, planet) {
  const sd = ensureSurfaceDelta(world, planet);
  if (!sd.sos || sd.sos[0] !== SOS_STATE.TRACKING) return { ok: false, reason: 'NoActiveSos' };
  sd.sos[0] = SOS_STATE.TRIGGERED;
  const branch = sd.sos[3];
  const fid = world.playerFactionId;

  if (branch === SOS_BRANCH.SURVIVOR_RESCUE) {
    // 分支 A：受困的先遣科考幸存者 -> 触发道德微抉择或直接救援入队
    sd.sos[0] = SOS_STATE.RESOLVED;
    credit(world, fid, 'credits', 400, 'sosRescue', planet.id, FLOW.LOOT);
    credit(world, fid, 'researchData', 35, 'sosRescue', planet.id, FLOW.LOOT);
    chronicle(world, 'discovery', `在行星 ${planet.name} 循求救信号救出受困先遣队员`, { planetId: planet.id });
    return { ok: true, branch: 'survivor_rescue', spawnAmbush: 0, reward: { credits: 400, researchData: 35 } };
  }
  if (branch === SOS_BRANCH.ALL_DEAD_SLATE) {
    // 分支 B：全员遇难留下的最后黑匣子绝笔 + 蓝图碎片 + 先驱者语素
    sd.sos[0] = SOS_STATE.RESOLVED;
    credit(world, fid, 'ancientData', 10, 'sosSlate', planet.id, FLOW.LOOT);
    credit(world, fid, 'alloy', 40, 'sosSlate', planet.id, FLOW.LOOT);
    return { ok: true, branch: 'all_dead_slate', spawnAmbush: 0, reward: { ancientData: 10, alloy: 40 } };
  }
  // 分支 C：海盗伪造信标设下的伏击陷阱！刷出 3 名海盗掠袭者 + 1 台重装机甲，击败后夺取高额战利品
  return {
    ok: true,
    branch: 'pirate_ambush',
    spawnAmbush: 4,
    ambushSpec: [{ kind: 'grunt', count: 3 }, { kind: 'heavy', count: 1 }],
  };
}

/**
 * 2. 行星生物与地质大勘探完整度计算（Planetary Survey Loop — 对标 Starfield）
 * 勘探要素权重：原生动物 (50%) + 奇异植物采样 (30%，需扫描至少 2 种) + 地质奇观/主遗迹勘测 (20%)
 */
export function computePlanetarySurveyPercent(planet) {
  if (!planet || planet.category === 'gasGiant') return 100;
  const sd = planet.surfaceDelta;
  const totalSpecies = getPlanetSpeciesCount(planet);
  if (totalSpecies === 0) return sd && sd.geoWonderScanned ? 100 : 0;

  let faunaMastered = 0;
  const targetFauna = Math.min(3, Math.max(1, totalSpecies - 4)); // 至少 3 种动物/物种达标
  for (let i = 0; i < totalSpecies; i++) {
    if (getSpeciesStage(planet, i) >= SPECIES_STAGE.IDENTIFIED) faunaMastered++;
  }
  const faunaRatio = clamp(faunaMastered / Math.max(1, targetFauna), 0, 1);

  // 统计 floraScannedMask 中置位的奇异植物种类（目标 2 种）
  let floraBits = 0;
  let mask = (sd && sd.floraScannedMask) || 0;
  while (mask) { floraBits += mask & 1; mask >>>= 1; }
  const floraRatio = clamp(floraBits / 2, 0, 1);

  // 地质奇观（1 处核心 POI 或巨石遗址完成勘测）
  const hasExploredPoi = Object.values(planet.pois || {}).some((p) => p.state === 'explored');
  const geoRatio = (sd && sd.geoWonderScanned) || hasExploredPoi ? 1.0 : 0.0;

  return Math.round((0.50 * faunaRatio + 0.30 * floraRatio + 0.20 * geoRatio) * 100);
}

  for (let i = sd.deployables.length - 1; i >= 0; i--) {
    const dep = sd.deployables[i]; // [id, kindIdx, x_dm, z_dm, hp, ttlSec_d10]
    dep[5] -= Math.round(dt * 10);
    if (dep[4] <= 0 || dep[5] <= 0) {
      sd.deployables.splice(i, 1);
      continue;
    }
    const dx = dep[2] / 10 - charState.x;
    const dz = dep[3] / 10 - charState.z;
    const dist = Math.hypot(dx, dz);

    if (dep[1] === 1 && dist <= FIELD_DEPLOYABLES.dep_thermal_pylon.radiusM) {
      // 热能发生器：庇护角色并回充护盾与燃料
      inThermalPylon = true;
      charState.shield = Math.min(charState.shieldMax, charState.shield + 6.0 * dt);
      charState.fuel = Math.min(100, charState.fuel + 10.0 * dt);
    } else if (dep[1] === 2) {
      // 便携哨兵机枪塔：自动索敌 24m 内最近敌人开火
      let bestFoe = null;
      let bestD = FIELD_DEPLOYABLES.dep_sentry_turret.rangeM;
      for (const e of enemies) {
        if (e.state === 'dead' || e.hp <= 0) continue;
        const ed = Math.hypot(e.x - dep[2] / 10, e.z - dep[3] / 10);
        if (ed <= bestD) { bestD = ed; bestFoe = e; }
      }
      if (bestFoe) {
        bestFoe.hp = Math.max(0, bestFoe.hp - FIELD_DEPLOYABLES.dep_sentry_turret.dps * dt);
        if (bestFoe.hp <= 0) bestFoe.state = 'dead';
      }
    } else if (dep[1] === 3) {
      activeDecoy = { x: dep[2] / 10, z: dep[3] / 10 };
    }
  }

  // 2) 生态感知与夜间/风暴狂暴化（掠食者感知半径 ×1.6，优先被诱饵弹吸引）
  for (const e of enemies) {
    if (e.state === 'dead') continue;
    e.effectiveAggroR = (e.baseAggroR || e.aggroR || 22) * (isNightOrStorm ? 1.6 : 1.0);
    if (activeDecoy) {
      e.overrideTarget = activeDecoy; // 诱饵弹强制拉走仇恨
    } else {
      e.overrideTarget = null;
    }
  }

  // 3) 警戒热度微分更新与星级跃迁
  const nearHostile = enemies.some((e) => e.state !== 'dead' && Math.hypot(e.x - charState.x, e.z - charState.z) < 42);
  const decayRate = (nearHostile && !activeDecoy ? 0 : 8.5) + (activeDecoy ? 25.0 : 0);
  const drillNoise = sd.extraction[0] === 1 ? 4.5 : 0; // 深核钻机运行噪音
  heat[0] = Math.round(clamp(heat[0] + (drillNoise - decayRate) * dt, 0, 1000));

  let newStar = 0;
  for (let s = 5; s >= 1; s--) {
    if (heat[0] >= HEAT_STAR_THRESHOLDS[s]) { newStar = s; break; }
  }
  const starEscalated = newStar > heat[1];
  heat[1] = newStar;

  return { heatPermille: heat[0], starLevel: heat[1], starEscalated, inThermalPylon, activeDecoy };
}
